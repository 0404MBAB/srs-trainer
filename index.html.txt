        const shuffleArray = (arr) => [...arr].sort(() => Math.random() - 0.5);

        // --- КОМПОНЕНТ ТАЙМЕРА ПОМОДОРО ---
        const PomodoroTimer = () => {
            const WORK_TIME = 25 * 60;
            const BREAK_TIME = 5 * 60;
            
            const [mode, setMode] = useState('work');
            const [timeLeft, setTimeLeft] = useState(WORK_TIME);
            const [isRunning, setIsRunning] = useState(false);
            
            const endTimeRef = useRef(null);

            useEffect(() => {
                let interval;
                if (isRunning) {
                    endTimeRef.current = Date.now() + timeLeft * 1000;
                    
                    interval = setInterval(() => {
                        const remaining = Math.round((endTimeRef.current - Date.now()) / 1000);
                        if (remaining <= 0) {
                            clearInterval(interval);
                            setTimeLeft(0);
                            setIsRunning(false);
                            if (mode === 'work') {
                                alert("Время отдыхать! Помодоро завершен.");
                                setMode('break');
                                setTimeLeft(BREAK_TIME);
                            } else {
                                alert("Отдых окончен! Пора за работу.");
                                setMode('work');
                                setTimeLeft(WORK_TIME);
                            }
                        } else {
                            setTimeLeft(remaining);
                        }
                    }, 500);
                }
                return () => clearInterval(interval);
            }, [isRunning, mode]);

            const toggleTimer = () => setIsRunning(!isRunning);
            const resetTimer = () => {
                setIsRunning(false);
                setTimeLeft(mode === 'work' ? WORK_TIME : BREAK_TIME);
            };
            const switchMode = (newMode) => {
                setIsRunning(false);
                setMode(newMode);
                setTimeLeft(newMode === 'work' ? WORK_TIME : BREAK_TIME);
            };

            const mins = String(Math.floor(timeLeft / 60)).padStart(2, '0');
            const secs = String(timeLeft % 60).padStart(2, '0');

            return (
                <div className="bg-white rounded-xl shadow-sm p-4 mb-6 text-center">
                    <div className="flex justify-center gap-2 mb-3">
                        <button onClick={() => switchMode('work')} className={`px-4 py-1 text-sm rounded-full font-medium transition-colors ${mode === 'work' ? 'bg-red-500 text-white' : 'bg-gray-100 text-gray-500'}`}>Работа (25м)</button>
                        <button onClick={() => switchMode('break')} className={`px-4 py-1 text-sm rounded-full font-medium transition-colors ${mode === 'break' ? 'bg-green-500 text-white' : 'bg-gray-100 text-gray-500'}`}>Отдых (5м)</button>
                    </div>
                    <div className="text-4xl font-bold font-mono text-gray-800 mb-4">{mins}:{secs}</div>
                    <div className="flex justify-center gap-3">
                        <button onClick={toggleTimer} className={`px-6 py-2 rounded-lg font-bold text-white transition-colors w-32 ${isRunning ? 'bg-yellow-500 hover:bg-yellow-600' : 'bg-indigo-600 hover:bg-indigo-700'}`}>
                            {isRunning ? 'Пауза' : 'Старт'}
                        </button>
                        <button onClick={resetTimer} className="px-4 py-2 rounded-lg bg-gray-200 text-gray-700 hover:bg-gray-300 font-medium">Сброс</button>
                    </div>
                </div>
            );
        };

                setQueue(orderMode === 'shuffle' ? shuffleArray(indices) : indices);
                setCurrentIndex(0);
                setIsFlipped(false);
            }, [orderMode]);

            const currentCard = ALL_CARDS[queue[currentIndex]] || null;

            const nextCard = (grade) => {
                if (queue.length === 0) return;
                
                let newQueue = [...queue];
                const currentId = newQueue[currentIndex];

                if (grade === 'hard') {
                    const insertPos = Math.min(currentIndex + 3, newQueue.length);
                    newQueue.splice(insertPos, 0, currentId);
                } else if (grade === 'normal') {
                    newQueue.push(currentId);
                }

                setIsFlipped(false);
                setQueue(newQueue);
                setCurrentIndex(prev => prev + 1);
            };

            const pickRandomCard = () => {
                const randomIndex = Math.floor(Math.random() * queue.length);
                setIsFlipped(false);
                setCurrentIndex(randomIndex);
            };

            if (currentIndex >= queue.length) {

                            <p className="text-gray-600 mb-6">Ты ответил на все вопросы в очереди.</p>

                    </div>

                            <div className="card-front bg-white rounded-2xl shadow-md p-6 flex flex-col cursor-pointer border border-gray-100" onClick={() => setIsFlipped(true)}>
                                <div className="text-xs font-bold text-indigo-500 uppercase tracking-wider mb-auto">{currentCard.cat}</div>
                                <div className="text-xl md:text-2xl font-bold text-gray-800 text-center leading-snug my-auto" dangerouslySetInnerHTML={{__html: currentCard.q}}>
                                </div>
                                <div className="text-sm text-gray-400 text-center mt-auto">Нажми, чтобы увидеть ответ</div>
                            </div>

                                    <div className="text-xs font-bold text-indigo-400 uppercase tracking-wider">{currentCard.cat}</div>
                                    <button 
                                        onClick={(e) => { e.stopPropagation(); setIsFlipped(false); }}
                                        className="text-xs bg-indigo-100 text-indigo-600 px-3 py-1 rounded-full font-bold hover:bg-indigo-200"
                                    >
                                        🔄 Вопрос

                                </div>
                            </div>
                        </div>
                    </div>

                    <div className={`w-full grid grid-cols-3 gap-3 transition-opacity duration-300 ${isFlipped ? 'opacity-100' : 'opacity-0 pointer-events-none'}`}>
                        <button onClick={() => nextCard('hard')} className="bg-red-100 text-red-700 py-3 rounded-xl font-bold hover:bg-red-200 active:scale-95 transition-all">
                            Сложно<br/><span className="text-xs font-normal opacity-75">Повторить скоро</span>
                        </button>

                            Нормально<br/><span className="text-xs font-normal opacity-75">В конец</span>
                        </button>

                            Легко<br/><span className="text-xs font-normal opacity-75">Знаю</span>

                </div>
            );
        };
